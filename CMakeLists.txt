cmake_minimum_required(VERSION 3.15)
project(azure_lane_pet CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

file(GLOB CORE_SRC src/core/*.cpp)
add_library(petcore STATIC ${CORE_SRC})
target_include_directories(petcore PUBLIC src)

add_executable(test_core tests/test_core.cpp)
target_link_libraries(test_core petcore)
enable_testing()
add_test(NAME core COMMAND test_core)

if(WIN32)
  file(GLOB WIN_SRC src/win/*.cpp)
  add_executable(azure_lane_pet WIN32 ${WIN_SRC} src/win/resource.rc)
  target_link_libraries(azure_lane_pet petcore gdiplus gdi32 user32 shell32 advapi32 ole32 winhttp)
  target_compile_definitions(azure_lane_pet PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  if(MSVC)
    target_compile_options(azure_lane_pet PRIVATE /utf-8)
    target_link_options(azure_lane_pet PRIVATE /ENTRY:wWinMainCRTStartup)
  else()
    target_link_options(azure_lane_pet PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
  add_custom_command(TARGET azure_lane_pet POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:azure_lane_pet>/assets)
endif()
